#include <stdio.h>
#include <pthread.h>
#include <unistd.h>
#define NUM_TASKS 10
#define NUM_THREADS 4
int tasks[NUM_TASKS];
void *process_tasks(void *arg) {
 int id = *(int *)arg;
 for (int i = id; i < NUM_TASKS; i += NUM_THREADS) {
 printf("Thread %d processing task %d\n", id, tasks[i]);
 sleep(1);
 }
 return NULL;
}
int main(void) {
 pthread_t threads[NUM_THREADS];
 int ids[NUM_THREADS];
 for (int i = 0; i < NUM_TASKS; i++)
 tasks[i] = i + 1;
 for (int i = 0; i < NUM_THREADS; i++) {
 ids[i] = i;
 pthread_create(&threads[i], NULL, process_tasks, &ids[i]);
 }
 for (int i = 0; i < NUM_THREADS; i++)
 pthread_join(threads[i], NULL);
 return 0;
}

Execution steps:
1. Save the code as p12.c
2. Compile: gcc p12.c -o p12 -pthread
3. Run: ./p12
