#include <mpi.h>
#include <stdio.h>
int main(int argc, char **argv) {
 int rank, size, name_len;
 char processor[MPI_MAX_PROCESSOR_NAME];
 MPI_Init(&argc, &argv);
 MPI_Comm_rank(MPI_COMM_WORLD, &rank);
 MPI_Comm_size(MPI_COMM_WORLD, &size);
 MPI_Get_processor_name(processor, &name_len);
 printf("Hello world from processor %s, rank %d out of %d processors\n",
 processor, rank, size);
 MPI_Finalize();
 return 0;
}

Execution steps:
1. Install MPICH: sudo apt update && sudo apt install mpich
2. Create a file named p1.c and paste the program code.
3. Compile: mpicc p1.c -o p1
4. Run five processes: mpirun -np 5 ./p1
