#include <stdio.h>
#include <omp.h>
#define SIZE 1000000
int array[SIZE];
int main(void) {
 int sum = 0;
 for (int i = 0; i < SIZE; i++)
 array[i] = i % 100;
 #pragma omp parallel for reduction(+:sum)
 for (int i = 0; i < SIZE; i++)
 sum += array[i];
 printf("Total Sum: %d\n", sum);
 return 0;
}

Execution steps:
1. Save the code as p9.c
2. Compile with OpenMP enabled: gcc p9.c -o p9 -fopenmp
3. Run: ./p9
