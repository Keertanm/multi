#include <stdio.h>
#include <pthread.h>
#define SIZE 100
#define THREADS 4
int data[SIZE];
int found_index = -1;
pthread_mutex_t lock = PTHREAD_MUTEX_INITIALIZER;
void *search(void *arg) {
 int start = *(int *)arg;
 for (int i = start; i < start + SIZE / THREADS; i++) {
 if (data[i] == 50) {
 pthread_mutex_lock(&lock);
 found_index = i;
 pthread_mutex_unlock(&lock);
 return NULL;
 }
 }
 return NULL;
}
int main(void) {
 pthread_t threads[THREADS];
 int starts[THREADS];
 for (int i = 0; i < SIZE; i++)
 data[i] = i + 1;
 for (int i = 0; i < THREADS; i++) {
 starts[i] = i * (SIZE / THREADS);
 pthread_create(&threads[i], NULL, search, &starts[i]);
 }
 for (int i = 0; i < THREADS; i++)
 pthread_join(threads[i], NULL);
 if (found_index != -1)
 printf("Number 50 found at index: %d\n", found_index);
 else
 printf("Number 50 not found.\n");
 pthread_mutex_destroy(&lock);
 return 0;
}

Execution steps:
1. Save the code as p5.c
2. Compile: gcc p5.c -o p5 -pthread
3. Run: ./p5
