#include <stdio.h>
#include <pthread.h>
#define SIZE 3
int A[SIZE][SIZE] = {{1, 2, 3}, {4, 5, 6}, {7, 8, 9}};
int B[SIZE][SIZE] = {{9, 8, 7}, {6, 5, 4}, {3, 2, 1}};
int C[SIZE][SIZE];
void *add_row(void *arg) {
 int row = *(int *)arg;
 printf("Thread ID: %lu is processing row %d\n",
 (unsigned long)pthread_self(), row);
 for (int col = 0; col < SIZE; col++)
 C[row][col] = A[row][col] + B[row][col];
 return NULL;
}
int main(void) {
 pthread_t threads[SIZE];
 int rows[SIZE];
 for (int i = 0; i < SIZE; i++) {
 rows[i] = i;
 pthread_create(&threads[i], NULL, add_row, &rows[i]);
 }
 for (int i = 0; i < SIZE; i++)
 pthread_join(threads[i], NULL);
 printf("Resultant Matrix:\n");
 for (int i = 0; i < SIZE; i++) {
 for (int j = 0; j < SIZE; j++)
 printf("%d ", C[i][j]);
 printf("\n");
 }
 return 0;
}

Execution steps:
1. Save the code as p3.c
2. Compile: gcc p3.c -o p3 -pthread
3. Run: ./p3
